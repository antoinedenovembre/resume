% ======================================================================
% =========================== MINIMAL STYLE ============================
% ======================================================================
% Loaded after style.tex by the "minimal" variants: one airy column,
% sans-serif font, gray small-caps section titles without rules, no icons.

\makeatletter
\usepackage[default]{sourcesanspro}
\usepackage{xstring}

\definecolor{minimalText}{HTML}{222222}
\definecolor{minimalMuted}{HTML}{7A7A7A}
\hypersetup{urlcolor=minimalText, linkcolor=minimalText}
\AtBeginDocument{\color{minimalText}}

\geometry{top=1.6 cm, bottom=1.4 cm, left=1.9 cm, right=1.9 cm}

% Section titles: small, letter-spaced uppercase in gray, no rule
\usepackage{microtype}
\titleformat{\section}
  {\needspace{4\baselineskip}\color{minimalMuted}\footnotesize\bfseries}
  {}{0pt}{\minimalspaced}
\titlespacing*{\section}{0pt}{2.2ex plus .3ex minus .2ex}{1.0ex plus .2ex minus .2ex}
\newcommand{\minimalspaced}[1]{\textls[150]{\MakeUppercase{#1}}}

\setlength{\parskip}{0pt}

% Entries: company + period on the first line, title + location on the second
\renewenvironment{resumeentry}[5][]{%
    \par\noindent
    {\bfseries #2}\hfill{\color{minimalMuted}\small #5}\par\noindent
    #3\hfill{\color{minimalMuted}\small #4}\par
}{%
    \par\vspace{0.9em}
}

\renewenvironment{resumehighlights}{%
    \begin{itemize}[
        label={\color{minimalMuted}\textendash},
        topsep=0.25em,
        partopsep=0pt,
        parsep=0pt,
        itemsep=0.15em,
        leftmargin=0.45 cm
    ]
}{%
    \end{itemize}
}

% Skills: gray label column on the left, value on the right
\newlength{\minimalskilllabel}
\setlength{\minimalskilllabel}{3.6 cm}
\renewenvironment{resumeskills}{}{}
\renewcommand{\resumeskill}[2]{%
    \par\noindent
    \parbox[t]{\minimalskilllabel}{\raggedright\leavevmode\color{minimalMuted}\StrDel{#1}{:}}%
    \parbox[t]{\linewidth-\minimalskilllabel}{\raggedright\leavevmode #2}\par\vspace{0.4em}
}

% Plain-text contact lines shared by both headers
\newcommand{\minimalsep}{\hspace{0.5em}{\color{minimalMuted}\textbullet}\hspace{0.5em}}
\newcommand{\minimalcontacts}{%
    {\color{minimalMuted}\small
    \PersonLocation\minimalsep
    \hrefWithoutArrow{tel:\PersonPhoneTel}{\color{minimalMuted}\PersonPhoneDisplay}\minimalsep
    \hrefWithoutArrow{mailto:\PersonEmail}{\color{minimalMuted}\PersonEmail}\\[0.15em]
    \hrefWithoutArrow{\PersonWebsiteURL}{\color{minimalMuted}\PersonWebsiteDisplay}\minimalsep
    \hrefWithoutArrow{\PersonLinkedinURL}{\color{minimalMuted}\PersonLinkedinDisplay}\minimalsep
    \hrefWithoutArrow{\PersonGithubURL}{\color{minimalMuted}github.com/\PersonGithubDisplay}}%
}
\makeatother
